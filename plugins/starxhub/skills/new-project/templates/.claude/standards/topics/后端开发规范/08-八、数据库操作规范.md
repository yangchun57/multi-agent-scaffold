### 8.1 SqlSugar 配置

```csharp
// Program.cs
builder.Services.AddScoped<ISqlSugarClient>(sp =>
{
    var configuration = sp.GetRequiredService<IConfiguration>();
    var connectionString = configuration.GetConnectionString("DefaultConnection");

    var db = new SqlSugarScope(new ConnectionConfig
    {
        ConnectionString = connectionString,
        DbType = DbType.MySql,
        IsAutoCloseConnection = true,
        InitKeyType = InitKeyType.Attribute
    },
    config =>
    {
        // 开发环境打印 SQL
#if DEBUG
        config.Aop.OnLogExecuting = (sql, pars) =>
        {
            Console.WriteLine($"[SQL] {sql}");
            if (pars != null && pars.Length > 0)
            {
                Console.WriteLine($"[Parameters] {string.Join(", ", pars.Select(p => $"{p.ParameterName}={p.Value}"))}");
            }
        };
#endif
    });

    return db;
});
```

### 8.2 查询规范

```csharp
// 基础查询
var list = await _db.Queryable<SysDepartment>()
    .Where(d => d.IsEnabled == true)
    .OrderBy(d => d.SortOrder)
    .ToListAsync();

// 条件查询（使用 WhereIF）
var list = await _db.Queryable<SysMeter>()
    .WhereIF(request.MeterType.HasValue, m => m.MeterType == request.MeterType)
    .WhereIF(request.DeptId.HasValue, m => m.DeptId == request.DeptId)
    .WhereIF(!string.IsNullOrEmpty(request.Keyword),
        m => m.MeterNo!.Contains(request.Keyword!) || m.Location.Contains(request.Keyword!))
    .ToListAsync();

// 关联查询
var list = await _db.Queryable<SysMeter, SysDepartment>(
    (m, d) => new JoinQueryInfos(JoinType.Left, m.DeptId == d.DeptId))
    .Where((m, d) => m.Status == MeterStatus.Active)
    .Select((m, d) => new MeterListDto
    {
        MeterId = m.MeterId,
        MeterNo = m.MeterNo,
        Location = m.Location,
        DeptName = d.DeptName
    })
    .ToListAsync();

// 分页查询
var total = await query.CountAsync();
var items = await query
    .Skip((pageIndex - 1) * pageSize)
    .Take(pageSize)
    .ToListAsync();
```

### 8.3 写入规范

```csharp
// 新增
var entity = new SysDepartment { /* ... */ };
await _db.Insertable(entity).ExecuteCommandAsync();

// 批量新增
await _db.Insertable(entities).ExecuteCommandAsync();

// 更新
await _db.Updateable(entity)
    .Where(d => d.DeptId == id)
    .ExecuteCommandAsync();

// 部分字段更新
await _db.Updateable<SysMeter>()
    .SetColumns(m => m.Status == newStatus)
    .Where(m => m.MeterId == id)
    .ExecuteCommandAsync();

// 删除
await _db.Deleteable<SysDepartment>()
    .Where(d => d.DeptId == id)
    .ExecuteCommandAsync();
```

### 8.4 事务规范

```csharp
// 方式一：使用事务方法
var result = await _db.Ado.UseTranAsync(async () =>
{
    // 操作一
    await _db.Insertable(entity1).ExecuteCommandAsync();
    // 操作二
    await _db.Updateable(entity2).ExecuteCommandAsync();
});

if (!result.IsSuccess)
{
    throw new BusinessException($"操作失败: {result.ErrorMessage}");
}

// 方式二：手动控制
try
{
    _db.Ado.BeginTran();

    // 业务操作...

    _db.Ado.CommitTran();
}
catch (Exception)
{
    _db.Ado.RollbackTran();
    throw;
}
```
