### 7.2 测试示例

```csharp
/// <summary>
/// 实体服务测试
/// </summary>
public class EntityServiceTests
{
    private readonly Mock<ISqlSugarClient> _mockDb;
    private readonly Mock<ILogger<EntityService>> _mockLogger;
    private readonly EntityService _service;

    public EntityServiceTests()
    {
        _mockDb = new Mock<ISqlSugarClient>();
        _mockLogger = new Mock<ILogger<EntityService>>();
        _service = new EntityService(_mockDb.Object, _mockLogger.Object);
    }

    [Fact]
    public async Task GetByIdAsync_ExistingId_ReturnsEntity()
    {
        // Arrange
        var expected = new Entity { Id = 1, Code = "0001", Name = "测试" };

        // Act
        var result = await _service.GetByIdAsync(1);

        // Assert
        Assert.NotNull(result);
        Assert.Equal("0001", result!.Code);
    }

    [Fact]
    public async Task CreateAsync_DuplicateCode_ReturnsFailure()
    {
        // Arrange
        var request = new EntityCreateRequest { Code = "0001", Name = "新实体" };

        // Act
        var (success, message, _) = await _service.CreateAsync(request);

        // Assert
        Assert.False(success);
        Assert.Equal("编码已存在", message);
    }
}
```

### 7.3 测试命名

- 格式：`{MethodName}_{Scenario}_{ExpectedResult}`
