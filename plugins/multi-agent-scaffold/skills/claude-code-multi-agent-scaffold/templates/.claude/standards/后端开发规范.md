## 三、分层架构规范

### 3.1 各层职责

| 层 | 命名空间 | 职责 | 禁止事项 |
|---|---------|------|---------|
| Controller | HNU.WaterElec.Api.Controllers | 参数接收与校验、权限控制、调用 Service、返回统一响应 | 禁止包含任何业务逻辑 |
| Service | HNU.WaterElec.Service | 所有业务逻辑、数据验证、事务管理、分摊/费用计算 | 禁止直接操作 HttpContext |
| Model | HNU.WaterElec.Model | 数据库实体映射、DTO 定义、枚举定义 | 禁止包含业务逻辑方法 |
| Common | HNU.WaterElec.Common | 公共工具类（JWT、Excel、统一返回格式等） | 禁止引用 Service 或 Model |

### 3.2 Controller 层规范

#### 3.2.1 基本结构

```csharp
/// <summary>
/// 部门管理控制器
/// </summary>
[ApiController]
[Route("api/[controller]")]
public class DepartmentController : ControllerBase
{
    private readonly IDepartmentService _departmentService;
    private readonly ILogger<DepartmentController> _logger;

    public DepartmentController(
        IDepartmentService departmentService,
        ILogger<DepartmentController> logger)
    {
        _departmentService = departmentService;
        _logger = logger;
    }

    /// <summary>
    /// 获取部门列表（分页）
    /// </summary>
    [HttpGet]
    [Authorize] // 所有已认证用户
    public async Task<ActionResult<ApiResult<PageResult<DepartmentDto>>>> GetList(
        [FromQuery] DepartmentQueryRequest request)
    {
        var result = await _departmentService.GetPagedListAsync(request);
        return Ok(ApiResult<PageResult<DepartmentDto>>.Success(result));
    }

    /// <summary>
    /// 获取全部部门（不分页）
    /// </summary>
    [HttpGet("all")]
    [Authorize]
    public async Task<ActionResult<ApiResult<List<DepartmentSimpleDto>>>> GetAll()
    {
        var result = await _departmentService.GetAllAsync();
        return Ok(ApiResult<List<DepartmentSimpleDto>>.Success(result));
    }

    /// <summary>
    /// 获取部门详情
    /// </summary>
    [HttpGet("{id}")]
    [Authorize]
    public async Task<ActionResult<ApiResult<DepartmentDto>>> GetById(long id)
    {
        var result = await _departmentService.GetByIdAsync(id);
        if (result == null)
        {
            return Ok(ApiResult<DepartmentDto>.Fail("部门不存在", 404));
        }
        return Ok(ApiResult<DepartmentDto>.Success(result));
    }

    /// <summary>
    /// 新增部门
    /// </summary>
    [HttpPost]
    [Authorize(Roles = "admin")]
    public async Task<ActionResult<ApiResult<DepartmentDto>>> Create(
        [FromBody] DepartmentCreateRequest request)
    {
        var result = await _departmentService.CreateAsync(request);
        return Ok(ApiResult<DepartmentDto>.Success(result, "创建成功"));
    }

    /// <summary>
    /// 编辑部门
    /// </summary>
    [HttpPut("{id}")]
    [Authorize(Roles = "admin")]
    public async Task<ActionResult<ApiResult<DepartmentDto>>> Update(
        long id, [FromBody] DepartmentUpdateRequest request)
    {
        request.DeptId = id;
        var result = await _departmentService.UpdateAsync(request);
        return Ok(ApiResult<DepartmentDto>.Success(result, "更新成功"));
    }

    /// <summary>
    /// 删除部门
    /// </summary>
    [HttpDelete("{id}")]
    [Authorize(Roles = "admin")]
    public async Task<ActionResult<ApiResult<object>>> Delete(long id)
    {
        await _departmentService.DeleteAsync(id);
        return Ok(ApiResult<object>.Success(null, "删除成功"));
    }
}
```

#### 3.2.2 Controller 规范要点

1. **路由规范**
   - 使用 `[ApiController]` 特性
   - 路由模板使用 `[Route("api/[controller]")]`
   - 子资源路由使用 RESTful 风格：`api/meters/{meterId}/sharings`

2. **权限控制**
   - 所有需要认证的接口添加 `[Authorize]`
   - 按角色控制使用 `[Authorize(Roles = "admin,finance")]`
   - 公开接口（如登录）不添加认证特性

3. **参数来源**
   - 查询参数使用 `[FromQuery]`
   - 路径参数使用路由模板参数
   - 请求体使用 `[FromBody]`
   - 表单数据使用 `[FromForm]`

4. **响应格式**
   - 统一使用 `ApiResult<T>` 包装响应
   - 成功：`ApiResult<T>.Success(data, message)`
   - 失败：`ApiResult<T>.Fail(message, code)`

#### 3.3.2 服务实现

```csharp
/// <summary>
/// 部门服务实现
/// </summary>
public class DepartmentService : IDepartmentService
{
    private readonly ISqlSugarClient _db;
    private readonly ILogger<DepartmentService> _logger;

    public DepartmentService(ISqlSugarClient db, ILogger<DepartmentService> logger)
    {
        _db = db;
        _logger = logger;
    }

    public async Task<PageResult<DepartmentDto>> GetPagedListAsync(DepartmentQueryRequest request)
    {
        var query = _db.Queryable<SysDepartment>()
            .WhereIF(!string.IsNullOrEmpty(request.Keyword),
                d => d.DeptCode.Contains(request.Keyword!) ||
                     d.DeptName.Contains(request.Keyword!))
            .WhereIF(request.IsEnabled.HasValue,
                d => d.IsEnabled == request.IsEnabled);

        var total = await query.CountAsync();

        var items = await query
            .OrderBy(d => d.SortOrder)
            .OrderBy(d => d.DeptId)
            .Skip((request.PageIndex - 1) * request.PageSize)
            .Take(request.PageSize)
            .Select<DepartmentDto>()
            .ToListAsync();

        return new PageResult<DepartmentDto>
        {
            Items = items,
            Total = total,
            PageIndex = request.PageIndex,
            PageSize = request.PageSize
        };
    }

    public async Task<DepartmentDto> CreateAsync(DepartmentCreateRequest request)
    {
        // 校验部门编码唯一性
        var exists = await _db.Queryable<SysDepartment>()
            .Where(d => d.DeptCode == request.DeptCode)
            .FirstAsync();

        if (exists != null)
        {
            throw new BusinessException("部门编码已存在");
        }

        var entity = new SysDepartment
        {
            DeptCode = request.DeptCode,
            DeptName = request.DeptName,
            ContactPerson = request.ContactPerson,
            ContactPhone = request.ContactPhone,
            SortOrder = request.SortOrder ?? 0,
            Remark = request.Remark,
            IsEnabled = true
        };

        await _db.Insertable(entity).ExecuteCommandAsync();

        _logger.LogInformation("创建部门: {DeptCode} - {DeptName}",
            entity.DeptCode, entity.DeptName);

        return await GetByIdAsync(entity.DeptId);
    }

    public async Task DeleteAsync(long id)
    {
        // 检查是否有仪表关联
        var meterCount = await _db.Queryable<SysMeter>()
            .Where(m => m.DeptId == id)
            .CountAsync();

        if (meterCount > 0)
        {
            throw new BusinessException("该部门下存在仪表，无法删除");
        }

        var result = await _db.Deleteable<SysDepartment>()
            .Where(d => d.DeptId == id)
            .ExecuteCommandAsync();

        if (result == 0)
        {
            throw new BusinessException("部门不存在");
        }

        _logger.LogInformation("删除部门: {DeptId}", id);
    }
}
```

#### 3.3.3 Service 规范要点

1. **依赖注入**
   - 通过构造函数注入 `ISqlSugarClient`
   - 注入所需的其他 Service（避免循环依赖）
   - 注入 `ILogger<T>` 用于日志记录

2. **事务管理**
   ```csharp
   public async Task<long> CalculateBillingAsync(string period, bool overwrite)
   {
       var result = await _db.Ado.UseTranAsync(async () =>
       {
           // 事务内的所有操作
           var readings = await GetConfirmedReadingsAsync(period);
           foreach (var reading in readings)
           {
               await CreateBillingAsync(reading);
           }
       });

       if (!result.IsSuccess)
       {
           throw new BusinessException($"费用计算失败: {result.ErrorMessage}");
       }

       return result.Data;
   }
   ```

3. **业务异常**
   - 抛出 `BusinessException` 表示业务校验失败
   - 包含明确的错误信息，便于前端展示

4. **日志记录**
   - 关键操作记录 Info 级别日志
   - 异常记录 Error 级别日志
   - 使用结构化日志格式

### 4.4 JSON 序列化与前后端字段映射规范

#### 4.4.1 JSON 序列化策略

ASP.NET Core 默认使用 `System.Text.Json`，将 C# `PascalCase` 属性自动序列化为 `camelCase` JSON 字段。项目必须保持此默认行为，禁止手动更改序列化策略。

```csharp
// Program.cs - 默认配置（无需额外设置）
builder.Services.AddControllers();
// 默认 PropertyNamingPolicy = JsonNamingPolicy.CamelCase
```

#### 4.4.2 DTO 字段命名约束

- 后端 DTO 属性名使用 `PascalCase`，前端 JSON 中自动变为 `camelCase`
- 响应 DTO 字段名应与数据库字段含义一一对应，不得随意改名
- 禁止在 DTO 中使用 `[JsonPropertyName("xxx")]` 手动覆盖字段名（除非有特殊业务需求并经过评审）

#### 4.4.3 字段映射示例

| 后端 C# 属性 (PascalCase) | 前端 JSON 字段 (camelCase) | 说明 |
|--------------------------|--------------------------|------|
| `RoomId` | `roomId` | 主键 |

### 6.2 全局异常过滤器

```csharp
/// <summary>
/// 全局异常处理过滤器
/// </summary>
public class GlobalExceptionFilter : IExceptionFilter
{
    private readonly ILogger<GlobalExceptionFilter> _logger;

    public GlobalExceptionFilter(ILogger<GlobalExceptionFilter> logger)
    {
        _logger = logger;
    }

    public void OnException(ExceptionContext context)
    {
        var exception = context.Exception;
        int code;
        string message;

        if (exception is BusinessException businessException)
        {
            code = businessException.Code;
            message = businessException.Message;
            _logger.LogWarning(exception, "业务异常: {Message}", message);
        }
        else
        {
            code = 500;
            message = "服务器内部错误";
            _logger.LogError(exception, "未处理异常: {Message}", exception.Message);
        }

        var result = new JsonResult(new ApiResult<object>
        {
            Code = code,
            Message = message,
            Data = null
        })
        {
            StatusCode = 200 // 始终返回 200，通过 code 区分
        };

        context.Result = result;
        context.ExceptionHandled = true;
    }
}
```

```csharp
/// <summary>
/// JWT 工具类
/// </summary>
public static class JwtHelper
{
    /// <summary>
    /// 生成 JWT Token
    /// </summary>
    public static string GenerateToken(long userId, string username, string roleCode,
        IConfiguration configuration)
    {
        var jwtSettings = configuration.GetSection("Jwt");
        var secretKey = jwtSettings["SecretKey"]!;
        var issuer = jwtSettings["Issuer"]!;
        var audience = jwtSettings["Audience"]!;
        var expirationMinutes = int.Parse(jwtSettings["ExpirationMinutes"]!);

        var claims = new[]
        {
            new Claim(ClaimTypes.NameIdentifier, userId.ToString()),
            new Claim(ClaimTypes.Name, username),
            new Claim(ClaimTypes.Role, roleCode),
            new Claim(JwtRegisteredClaimNames.Jti, Guid.NewGuid().ToString())
        };

        var key = new SymmetricSecurityKey(Encoding.UTF8.GetBytes(secretKey));
        var credentials = new SigningCredentials(key, SecurityAlgorithms.HmacSha256);

        var token = new JwtSecurityToken(
            issuer: issuer,
            audience: audience,
            claims: claims,
            expires: DateTime.UtcNow.AddMinutes(expirationMinutes),
            signingCredentials: credentials
        );

        return new JwtSecurityTokenHandler().WriteToken(token);
    }
}
```

```csharp
/// 导入错误
/// </summary>
public class ImportError
{
    /// <summary>
    /// 行号
    /// </summary>

    /// <summary>
    /// 错误原因
    /// </summary>

/// <summary>
/// Excel 导入服务示例
/// </summary>
public async Task<ImportResult> ImportMetersAsync(IFormFile file)
{
    var result = new ImportResult();

    using var stream = file.OpenReadStream();
    var rows = stream.Query<MeterImportRow>(startCell: "A2").ToList();

    for (int i = 0; i < rows.Count; i++)
    {
        var row = rows[i];
        var rowNum = i + 2; // Excel 行号从 2 开始（第1行是表头）

        try
        {
            // 校验数据
            if (string.IsNullOrEmpty(row.DeptCode))
            {
                result.Errors.Add(new ImportError { Row = rowNum, Reason = "部门编码不能为空" });
                result.FailCount++;
                continue;
            }

            // 查找部门
            var dept = await _db.Queryable<SysDepartment>()
                .Where(d => d.DeptCode == row.DeptCode)
                .FirstAsync();

            if (dept == null)
            {
                result.Errors.Add(new ImportError { Row = rowNum, Reason = $"部门不存在: {row.DeptCode}" });
                result.FailCount++;
                continue;
            }

            // 创建实体
            var entity = new SysMeter
            {
                DeptId = dept.DeptId,
                MeterType = row.MeterType == "电表" ? MeterType.Electric : MeterType.Water,
                SeqNo = row.SeqNo,
                MeterNo = row.MeterNo,
                Location = row.Location,
                Status = MeterStatus.Active
            };

            await _db.Insertable(entity).ExecuteCommandAsync();
            result.SuccessCount++;
        }
        catch (Exception ex)
        {
            result.Errors.Add(new ImportError { Row = rowNum, Reason = ex.Message });
            result.FailCount++;
        }
    }

    return result;
}
```
