### 12.2 测试示例

```csharp
/// <summary>
/// 部门服务测试
/// </summary>
public class DepartmentServiceTests
{
    private readonly Mock<ISqlSugarClient> _mockDb;
    private readonly Mock<ILogger<DepartmentService>> _mockLogger;
    private readonly DepartmentService _service;

    public DepartmentServiceTests()
    {
        _mockDb = new Mock<ISqlSugarClient>();
        _mockLogger = new Mock<ILogger<DepartmentService>>();
        _service = new DepartmentService(_mockDb.Object, _mockLogger.Object);
    }

    [Fact]
    public async Task GetByIdAsync_ExistingId_ReturnsDepartment()
    {
        // Arrange
        var expected = new SysDepartment { DeptId = 1, DeptCode = "0001", DeptName = "测试部门" };

        // Act
        var result = await _service.GetByIdAsync(1);

        // Assert
        Assert.NotNull(result);
        Assert.Equal("0001", result!.DeptCode);
    }

    [Fact]
    public async Task CreateAsync_DuplicateCode_ThrowsBusinessException()
    {
        // Arrange
        var request = new DepartmentCreateRequest { DeptCode = "0001", DeptName = "新部门" };

        // Act & Assert
        await Assert.ThrowsAsync<BusinessException>(() => _service.CreateAsync(request));
    }
}
```
