# 前端开发规范 · 二、组件开发规范

> 拆分自《前端开发规范》第 2 节。需要全量上下文时读原文件。


### 2.1 组件结构

使用 Vue 3 Composition API + `<script setup>` 语法：

```vue
<template>
  <div class="component-name">
    <!-- 模板内容 -->
  </div>
</template>

<script setup lang="ts">
import { ref, onMounted, watch } from 'vue'
// 导入 API
import { getItems } from '@/api/module'
// 导入类型
import type { Item } from '@/types/module.d'
// 导入组件
import SubComponent from '@/components/SubComponent.vue'

// Props 定义
const props = defineProps<{
  modelValue?: number | null
  disabled?: boolean
}>()

// Emits 定义
const emit = defineEmits<{
  'update:modelValue': [value: number | null]
  change: [value: number | null]
}>()

// 响应式状态
const loading = ref(false)
const items = ref<Item[]>([])

// 方法
async function loadData() {
  loading.value = true
  try {
    const res = await getItems()
    if (res.code === 200) {
      items.value = res.data
    }
  } finally {
    loading.value = false
  }
}

// 生命周期
onMounted(loadData)
</script>

<style scoped>
.component-name {
  /* 样式 */
}
</style>
```

### 2.2 Props 和 Emits 规范

**Props 定义**：使用 TypeScript 泛型语法

```typescript
// 推荐
defineProps<{
  modelValue?: number | null
  disabled?: boolean
  placeholder?: string
}>()

// 不推荐（使用运行时声明）
defineProps({
  modelValue: [Number, null],
  disabled: Boolean,
})
```

**Emits 定义**：使用类型化语法

```typescript
const emit = defineEmits<{
  'update:modelValue': [value: number | null]
  change: [value: number | null]
}>()
```

### 2.3 选择器组件规范

选择器组件应遵循以下规范：

```vue
<template>
  <el-select
    :model-value="modelValue"
    :placeholder="placeholder"
    :clearable="clearable"
    :disabled="disabled"
    filterable
    @update:model-value="handleChange"
  >
    <el-option
      v-for="item in options"
      :key="item.id"
      :label="item.name"
      :value="item.id"
    />
  </el-select>
</template>

<script setup lang="ts">
import { ref, watch, onMounted } from 'vue'
import { getOptions } from '@/api/module'
import type { Option } from '@/types/module.d'

const props = defineProps<{
  modelValue?: string | null
  parentId?: string | null    // 父级ID（雪花 ID），用于级联
  clearable?: boolean
  disabled?: boolean
  placeholder?: string
}>()

const emit = defineEmits<{
  'update:modelValue': [value: number | null]
  change: [value: number | null]
}>()

const options = ref<Option[]>([])

// 父级变化时清空选择并重新加载
watch(() => props.parentId, () => {
  options.value = []
  emit('update:modelValue', null)
  emit('change', null)
  loadOptions()
}, { immediate: true })

async function loadOptions() {
  if (!props.parentId) return
  const res = await getOptions({ parentId: props.parentId })
  if (res.code === 200) options.value = res.data
}

function handleChange(val: number | null) {
  emit('update:modelValue', val)
  emit('change', val)
}
</script>
```

### 2.4 列表页面规范

列表页面应包含以下标准结构：

```vue
<template>
  <div class="page-name">
    <el-card shadow="never">
      <!-- 卡片标题 -->
      <template #header>
        <div class="card-header">
          <span>页面标题</span>
          <el-button type="primary" size="small" @click="openDialog()">
            新增
          </el-button>
        </div>
      </template>

      <!-- 搜索表单 -->
      <el-form :inline="true" :model="query" class="search-form">
        <el-form-item label="筛选条件">
          <Selector v-model="query.field" style="width: 160px" />
        </el-form-item>
        <el-form-item>
          <el-button type="primary" @click="loadData">查询</el-button>
        </el-form-item>
      </el-form>

      <!-- 数据表格 -->
      <el-table :data="tableData" border style="width: 100%">
        <!-- 列定义 -->
      </el-table>

      <!-- 分页 -->
      <el-pagination
        v-if="total > 0"
        class="pagination"
        :current-page="query.pageIndex"
        :page-size="query.pageSize"
        :total="total"
        layout="total, prev, pager, next"
        @current-change="handlePageChange"
      />
    </el-card>

    <!-- 编辑弹窗 -->
    <el-dialog v-model="dialogVisible" :title="form.id ? '编辑' : '新增'">
      <!-- 表单内容 -->
    </el-dialog>
  </div>
</template>

<script setup lang="ts">
import { ref, onMounted, watch } from 'vue'
import { ElMessage, ElMessageBox } from 'element-plus'
// 导入...

// 查询参数
const query = ref<QueryType>({ pageIndex: 1, pageSize: 20 })
// 表格数据
const tableData = ref<Item[]>([])
const total = ref(0)
// 弹窗状态
const dialogVisible = ref(false)
const saving = ref(false)
const form = ref<FormType>({})

// 加载数据
async function loadData() {
  const res = await getItems(query.value)
  if (res.code === 200) {
    tableData.value = res.data.items
    total.value = res.data.total
  }
}

// 分页
function handlePageChange(page: number) {
  query.value.pageIndex = page
  loadData()
}

// 打开弹窗
function openDialog(item?: Item) {
  if (item) {
    form.value = { ...item }
  } else {
    form.value = {}
  }
  dialogVisible.value = true
}

// 保存
async function handleSave() {
  saving.value = true
  try {
    if (form.value.id) {
      await updateItem(form.value.id, form.value)
    } else {
      await createItem(form.value)
    }
    ElMessage.success('保存成功')
    dialogVisible.value = false
    await loadData()
  } finally {
    saving.value = false
  }
}

// 删除
async function handleDelete(item: Item) {
  await ElMessageBox.confirm(`确定删除「${item.name}」？`, '提示', { type: 'warning' })
  const res = await deleteItem(item.id)
  if (res.code === 200) {
    ElMessage.success('删除成功')
    await loadData()
  } else {
    ElMessage.error(res.message)
  }
}

onMounted(loadData)
</script>

<style scoped>
.card-header {
  display: flex;
  justify-content: space-between;
  align-items: center;
}
.search-form {
  margin-bottom: 16px;
}
.pagination {
  margin-top: 16px;
  justify-content: flex-end;
}
</style>
```
