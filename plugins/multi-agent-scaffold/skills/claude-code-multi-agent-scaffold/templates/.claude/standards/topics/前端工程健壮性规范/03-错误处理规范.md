# 前端工程健壮性规范 · 错误处理规范

> 对应问题：H-10（catch 块混淆用户取消与真实错误）、H-11（错误仅 console.log，无用户反馈）

## 强制规则

### 1. 删除操作必须区分用户取消和真实错误

```typescript
// ❌ 错误：catch 块吞掉所有错误
async function handleDelete(id: string) {
  try {
    await ElMessageBox.confirm('确定删除？', '提示', { type: 'warning' })
    await deleteItem(id)
    ElMessage.success('删除成功')
  } catch {
    // 错误！用户取消和真实错误都被吞掉
    console.log('删除取消')
  }
}

// ✅ 正确：区分用户取消和真实错误
async function handleDelete(id: string) {
  try {
    await ElMessageBox.confirm('确定删除？', '提示', { type: 'warning' })
    await deleteItem(id)
    ElMessage.success('删除成功')
    await loadData()
  } catch (error) {
    // 判断是否为用户取消
    if (error === 'cancel' || error?.toString().includes('cancel')) {
      return  // 用户取消，不处理
    }
    
    // 真实错误，给用户反馈
    console.error('删除失败:', error)
    ElMessage.error('删除失败，请重试')
  }
}
```

### 2. API 调用错误必须给用户反馈

```typescript
// ❌ 错误：只 console.log，用户不知道发生了什么
async function loadData() {
  try {
    const res = await getItems()
    tableData.value = res.data
  } catch (error) {
    console.log('loadData error:', error)  // 错误！用户无感知
  }
}

// ✅ 正确：给用户明确的错误提示
async function loadData() {
  loading.value = true
  try {
    const res = await getItems()
    if (res.code === 200) {
      tableData.value = res.data.items
      total.value = res.data.total
    } else {
      ElMessage.error(res.message || '加载失败')
    }
  } catch (error) {
    console.error('loadData error:', error)
    ElMessage.error('网络错误，请稍后重试')
  } finally {
    loading.value = false
  }
}
```

### 3. 表单提交错误处理

```typescript
// ✅ 完整示例：表单提交错误处理
async function handleSubmit() {
  // 1. 表单验证
  const valid = await formRef.value?.validate().catch(() => false)
  if (!valid) {
    ElMessage.warning('请检查表单填写')
    return
  }
  
  // 2. 提交数据
  submitting.value = true
  try {
    const res = formData.value.id 
      ? await updateItem(formData.value.id, formData.value)
      : await createItem(formData.value)
    
    if (res.code === 200) {
      ElMessage.success(formData.value.id ? '更新成功' : '创建成功')
      dialogVisible.value = false
      await loadData()
    } else {
      ElMessage.error(res.message || '操作失败')
    }
  } catch (error) {
    // 3. 网络错误或服务器错误
    console.error('提交失败:', error)
    
    if (axios.isAxiosError(error)) {
      if (error.response?.status === 409) {
        ElMessage.error('数据已被修改，请刷新后重试')
      } else if (error.response?.status === 400) {
        ElMessage.error(error.response.data?.message || '请求参数错误')
      } else {
        ElMessage.error('网络错误，请稍后重试')
      }
    } else {
      ElMessage.error('未知错误，请重试')
    }
  } finally {
    submitting.value = false
  }
}
```

### 4. 封装统一的错误处理工具

```typescript
// utils/error-handler.ts
import { ElMessage } from 'element-plus'
import axios from 'axios'

export function handleError(error: unknown, context?: string) {
  console.error(context ? `${context} error:` : 'Error:', error)
  
  // 用户取消操作
  if (error === 'cancel' || error?.toString().includes('cancel')) {
    return
  }
  
  // Axios 错误
  if (axios.isAxiosError(error)) {
    if (!error.response) {
      ElMessage.error('网络连接失败，请检查网络')
      return
    }
    
    const status = error.response.status
    const message = error.response.data?.message
    
    switch (status) {
      case 400:
        ElMessage.error(message || '请求参数错误')
        break
      case 401:
        ElMessage.error('登录已过期，请重新登录')
        // 跳转到登录页
        break
      case 403:
        ElMessage.error('没有权限执行此操作')
        break
      case 404:
        ElMessage.error('请求的资源不存在')
        break
      case 409:
        ElMessage.error('数据冲突，请刷新后重试')
        break
      case 500:
        ElMessage.error('服务器内部错误')
        break
      default:
        ElMessage.error(message || `请求失败 (${status})`)
    }
    return
  }
  
  // 其他错误
  ElMessage.error('操作失败，请稍后重试')
}

// 使用
async function loadData() {
  try {
    const res = await getItems()
    tableData.value = res.data
  } catch (error) {
    handleError(error, 'loadData')
  }
}
```
