### 4.1 命名规范

| 类型 | 命名规则 | 示例 |
|------|---------|------|
| 页面组件 | `{PageName}.vue` | `MeterList.vue`, `FamilyBillingList.vue` |
| 公共组件 | `{Component}.vue` | `CampusSelector.vue`, `DeptSelector.vue` |
| API 文件 | `{module}.ts` | `family-base.ts`, `meter.ts` |
| 类型文件 | `{module}.d.ts` | `family.d.ts`, `api.d.ts` |
| 变量/函数 | camelCase | `loading`, `tableData`, `handleSave` |
| 接口/类型 | PascalCase | `ApiResult`, `PageResult`, `EntityDto` |
| 常量 | UPPER_SNAKE_CASE | `DEFAULT_PAGE_SIZE`, `API_BASE_URL` |

### 4.4 列表页面规范

```vue
<template>
  <div class="page-name">
    <el-card shadow="never">
      <template #header>
        <div class="card-header">
          <span>页面标题</span>
          <el-button type="primary" size="small" @click="openDialog()">
            新增
          </el-button>
        </div>
      </template>

      <!-- 搜索表单 -->
      <el-form :inline="true" :model="query" class="search-form">
        <el-form-item label="筛选条件">
          <Selector v-model="query.field" style="width: 160px" />
        </el-form-item>
        <el-form-item>
          <el-button type="primary" @click="loadData">查询</el-button>
        </el-form-item>
      </el-form>

      <!-- 数据表格 -->
      <el-table v-loading="loading" :data="tableData" border style="width: 100%">
        <!-- 列定义 -->
      </el-table>

      <!-- 分页 -->
      <el-pagination
        v-if="total > 0"
        class="pagination"
        :current-page="query.pageIndex"
        :page-size="query.pageSize"
        :total="total"
        layout="total, prev, pager, next"
        @current-change="handlePageChange"
      />
    </el-card>

    <!-- 编辑弹窗 -->
    <el-dialog v-model="dialogVisible" :title="form.id ? '编辑' : '新增'">
      <!-- 表单内容 -->
    </el-dialog>
  </div>
</template>

<script setup lang="ts">
import { ref, onMounted } from 'vue'
import { ElMessage, ElMessageBox } from 'element-plus'
import type { FormInstance } from 'element-plus'
import { getItems, createItem, updateItem, deleteItem } from '@/api/module'
import type { Item, QueryType, FormType } from '@/types/module.d'

const query = ref<QueryType>({ pageIndex: 1, pageSize: 20 })
const tableData = ref<Item[]>([])
const total = ref(0)
const loading = ref(false)
const dialogVisible = ref(false)
const saving = ref(false)
const form = ref<FormType>({})
const formRef = ref<FormInstance>()

async function loadData() {
  loading.value = true
  try {
    const res = await getItems(query.value)
    if (res.code === 200) {
      tableData.value = res.data.items
      total.value = res.data.total
    }
  } finally {
    loading.value = false
  }
}

function handlePageChange(page: number) {
  query.value.pageIndex = page
  loadData()
}

function openDialog(item?: Item) {
  if (item) {
    form.value = { ...item }
  } else {
    form.value = {}
  }
  dialogVisible.value = true
}

async function handleSave() {
  const valid = await formRef.value?.validate()
  if (!valid) return

  saving.value = true
  try {
    if (form.value.id) {
      await updateItem(form.value.id, form.value)
    } else {
      await createItem(form.value)
    }
    ElMessage.success('保存成功')
    dialogVisible.value = false
    await loadData()
  } finally {
    saving.value = false
  }
}

async function handleDelete(item: Item) {
  await ElMessageBox.confirm(`确定删除「${item.name}」？`, '提示', { type: 'warning' })
  const res = await deleteItem(item.id)
  if (res.code === 200) {
    ElMessage.success('删除成功')
    await loadData()
  } else {
    ElMessage.error(res.message)
  }
}

onMounted(loadData)
</script>

<style scoped>
.card-header {
  display: flex;
  justify-content: space-between;
  align-items: center;
}
.search-form {
  margin-bottom: 16px;
}
.pagination {
  margin-top: 16px;
  justify-content: flex-end;
}
</style>
```

### 4.5 TypeScript 类型规范

#### 4.5.1 实体类型定义

```typescript
// 完整实体
export interface Entity {
  id: string          // 雪花 ID，必须用 string（防止精度丢失）
  code: string
  name: string
  sortOrder: number
  isEnabled: boolean
  remark?: string
  createdAt: string
  updatedAt: string
}

// 表单类型
export interface EntityForm {
  id?: string         // 雪花 ID
  code: string
  name: string
  sortOrder?: number
  remark?: string
}

// 查询参数
export interface EntityQuery extends PageQuery {
  keyword?: string
  status?: number | null
}

// 通用分页参数
export interface PageQuery {
  pageIndex: number
  pageSize: number
}

// 通用响应
export interface ApiResult<T = any> {
  code: number
  message: string
  data: T
  timestamp: string
}

// 分页响应
export interface PageResult<T = any> {
  items: T[]
  total: number
  pageIndex: number
  pageSize: number
}
```

### 4.6 API 接口规范

```typescript
// 按模块划分 API 文件
// api/entity.ts
import request from './request'
import type { ApiResult, PageResult } from '@/types/api.d'
import type { Entity, EntityForm, EntityQuery } from '@/types/entity.d'

export function getEntities(params: EntityQuery): Promise<ApiResult<PageResult<Entity>>> {
  return request.get('/entity', { params })
}

export function getEntityById(id: string): Promise<ApiResult<Entity>> {
  return request.get(`/entity/${id}`)
}

export function createEntity(data: EntityForm): Promise<ApiResult<Entity>> {
  return request.post('/entity', data)
}

export function updateEntity(id: string, data: EntityForm): Promise<ApiResult<Entity>> {
  return request.put(`/entity/${id}`, data)
}

export function deleteEntity(id: string): Promise<ApiResult<null>> {
  return request.delete(`/entity/${id}`)
}
```
